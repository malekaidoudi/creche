Critères de validation:
- Minimum 8 caractères
- Au moins une majuscule
- Au moins une minuscule
- Au moins un chiffre
- Confirmation identique

API utilisée:
POST /api/auth/create-password

Body:
```json
{
  "token": "abc123...",
  "email": "parent@example.com",
  "password": "MotDePasse123"
}
```

Réponse:
```json
{
  "success": true,
  "message": "Compte créé avec succès",
  "token": "jwt_token_here",
  "user": {
    "id": 1,
    "email": "parent@example.com",
    "first_name": "Ahmed",
    "last_name": "Ben Ali",
    "role": "parent"
  }
}
```

Flow:
1. Parent reçoit e-mail avec lien
2. Clique sur le lien → /create-password?token=xxx&email=yyy
3. Entre son mot de passe (2x)
4. Validation en temps réel
5. Soumission → API
6. Token JWT sauvegardé
7. Redirection vers /dashboard

📊 INTÉGRATION DANS LA PAGE DES INSCRIPTIONS
═══════════════════════════════════════════════════════════════════════════════

Pour utiliser le modal d'approbation dans EnrollmentsPage.jsx:

```jsx
import { useState } from 'react';
import ApproveEnrollmentModal from '../../components/modals/ApproveEnrollmentModal';

const EnrollmentsPage = () => {
  const [showApproveModal, setShowApproveModal] = useState(false);
  const [selectedEnrollment, setSelectedEnrollment] = useState(null);

  const handleApproveClick = (enrollment) => {
    setSelectedEnrollment(enrollment);
    setShowApproveModal(true);
  };

  const handleApprove = async (enrollmentId, appointmentDate) => {
    try {
      const token = localStorage.getItem('token');
      const response = await fetch(`/api/enrollments/${enrollmentId}/approve`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ appointment_date: appointmentDate })
      });

      if (!response.ok) {
        throw new Error('Erreur lors de l\'approbation');
      }

      // Rafraîchir la liste
      await fetchEnrollments();
      
      // Fermer le modal
      setShowApproveModal(false);
      
    } catch (error) {
      console.error('Erreur:', error);
      throw error; // Le modal affichera l'erreur
    }
  };

  return (
    <div>
      {/* Liste des inscriptions */}
      {enrollments.map(enrollment => (
        <div key={enrollment.id}>
          {/* ... */}
          <button onClick={() => handleApproveClick(enrollment)}>
            Approuver
          </button>
        </div>
      ))}

      {/* Modal d'approbation */}
      <ApproveEnrollmentModal
        isOpen={showApproveModal}
        onClose={() => setShowApproveModal(false)}
        enrollment={selectedEnrollment}
        onApprove={handleApprove}
      />
    </div>
  );
};
```
