# Voir les derniers e-mails envoyés
SELECT * FROM email_logs ORDER BY created_at DESC LIMIT 10;

# Voir les e-mails échoués
SELECT * FROM email_logs WHERE status = 'failed' ORDER BY created_at DESC;

# Statistiques par type d'e-mail
SELECT 
  email_type, 
  COUNT(*) as total,
  SUM(CASE WHEN status = 'sent' THEN 1 ELSE 0 END) as sent,
  SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) as failed
FROM email_logs
GROUP BY email_type;

# E-mails envoyés aujourd'hui
SELECT * FROM email_logs 
WHERE DATE(created_at) = CURRENT_DATE
ORDER BY created_at DESC;

# Taux de succès des 7 derniers jours
SELECT 
  ROUND(
    100.0 * SUM(CASE WHEN status = 'sent' THEN 1 ELSE 0 END) / COUNT(*),
    2
  ) as success_rate
FROM email_logs
WHERE created_at > NOW() - INTERVAL '7 days';

# Messages de contact non lus
SELECT * FROM contact_messages 
WHERE status = 'new' 
ORDER BY created_at DESC;

# Tous les messages de contact
SELECT * FROM contact_messages 
ORDER BY created_at DESC 
LIMIT 20;

📊 MONITORING EN PRODUCTION
═══════════════════════════════════════════════════════════════════════════════

# Vérifier les e-mails des dernières 24h
SELECT 
  email_type,
  status,
  COUNT(*) as count
FROM email_logs
WHERE created_at > NOW() - INTERVAL '24 hours'
GROUP BY email_type, status
ORDER BY email_type, status;

# Vérifier les erreurs récentes
SELECT 
  email_type,
  recipient_email,
  error_message,
  created_at
FROM email_logs
WHERE status = 'failed'
  AND created_at > NOW() - INTERVAL '7 days'
ORDER BY created_at DESC;

# Statistiques hebdomadaires
SELECT 
  DATE(created_at) as date,
  COUNT(*) as total,
  SUM(CASE WHEN status = 'sent' THEN 1 ELSE 0 END) as sent,
  SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) as failed
FROM email_logs
WHERE created_at > NOW() - INTERVAL '7 days'
GROUP BY DATE(created_at)
ORDER BY date DESC;

🧹 MAINTENANCE
═══════════════════════════════════════════════════════════════════════════════

# Nettoyer les logs de plus de 90 jours
DELETE FROM email_logs 
WHERE created_at < NOW() - INTERVAL '90 days';

# Compter les e-mails par mois
SELECT 
  DATE_TRUNC('month', created_at) as month,
  COUNT(*) as total
FROM email_logs
GROUP BY DATE_TRUNC('month', created_at)
ORDER BY month DESC;

# Marquer les messages de contact comme lus
UPDATE contact_messages 
SET status = 'read' 
WHERE id = 123;

# Marquer un message comme répondu
UPDATE contact_messages 
SET 
  status = 'responded',
  responded_at = NOW(),
  responded_by = 1
WHERE id = 123;
